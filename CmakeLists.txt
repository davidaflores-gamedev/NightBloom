#------------------------------------------------------------------------------
# NightBloom_Engine/CMakeLists.txt (ROOT)
#
# Root build configuration - builds both Engine and Editor
#------------------------------------------------------------------------------
cmake_minimum_required(VERSION 3.20)
project(NightbloomWorkspace VERSION 0.1.0 LANGUAGES CXX)

#------------------------------------------------------------------------------
# Global C++ settings (applies to ALL subprojects)
#------------------------------------------------------------------------------
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

#------------------------------------------------------------------------------
# Shared CMake helpers. NightbloomShaders defines nightbloom_compile_shaders(),
# used by any exe target (Editor now, Game runtime later) to deploy the
# engine-owned shaders. Included before the targets so subdirectories see it.
#------------------------------------------------------------------------------
include(${CMAKE_SOURCE_DIR}/cmake/NightbloomShaders.cmake)

#------------------------------------------------------------------------------
# Build Engine (includes ThirdParty automatically)
#------------------------------------------------------------------------------
add_subdirectory(NightBloom)

#------------------------------------------------------------------------------
# Build Editor (links against engine)
#------------------------------------------------------------------------------
add_subdirectory(Editor)

#------------------------------------------------------------------------------
# Build standalone Game runtime (Milestone 2 — links engine, no editor UI)
#------------------------------------------------------------------------------
add_subdirectory(Game)

#------------------------------------------------------------------------------
# Packaging (Milestone 3 — see .claude/PACKAGING_ROADMAP.md): assemble a
# distributable, self-contained NightbloomGame zip. The heavy lifting (staging
# exe + Shaders/ + Assets/ + the project with project-relative scene paths, then
# zipping) lives in Packaging/PackageGame.ps1 — a package script is the roadmap's
# accepted alternative to install()/CPack, and it also migrates legacy absolute
# asset paths, which pure CMake can't. This target just builds the Game and runs
# the script for the current config. Not part of ALL_BUILD.
#   cmake --build build --target package --config Release
#------------------------------------------------------------------------------
if(WIN32)
	add_custom_target(package
		COMMAND powershell -ExecutionPolicy Bypass -File
			"${CMAKE_SOURCE_DIR}/Packaging/PackageGame.ps1" -Config $<CONFIG> -SkipBuild
		WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
		COMMENT "Packaging NightbloomGame ($<CONFIG>) -> dist/"
		VERBATIM)
	add_dependencies(package NightbloomGame)
endif()

#------------------------------------------------------------------------------
# Set Editor as the startup project in Visual Studio
#------------------------------------------------------------------------------
set_property(DIRECTORY ${CMAKE_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT NightbloomEditor)
